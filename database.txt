-- 1. Bảng Quyền (Roles)
CREATE TABLE Roles (
    id INT PRIMARY KEY IDENTITY(1,1),
    role_name NVARCHAR(50) NOT NULL UNIQUE
);

-- 2. Bảng Người dùng (Users)
CREATE TABLE Users (
    id INT PRIMARY KEY IDENTITY(1,1),
    username NVARCHAR(50) NOT NULL UNIQUE,
    password_hash NVARCHAR(255) NOT NULL,
    role_id INT FOREIGN KEY REFERENCES Roles(id),
    created_at DATETIME DEFAULT GETDATE()
);

-- 3. Bảng Danh mục sản phẩm (Categories)
CREATE TABLE Categories (
    id INT PRIMARY KEY IDENTITY(1,1),
    name NVARCHAR(100) NOT NULL UNIQUE
);

-- 4. Bảng Sản phẩm (Products)
CREATE TABLE Products (
    id INT PRIMARY KEY IDENTITY(1,1),
    sku NVARCHAR(50) NOT NULL UNIQUE,
    name NVARCHAR(255) NOT NULL,
    category_id INT FOREIGN KEY REFERENCES Categories(id),
    min_stock INT DEFAULT 0,
    description NVARCHAR(MAX),
    created_at DATETIME DEFAULT GETDATE()
);

-- 5. Bảng Kho hàng (Warehouses)
CREATE TABLE Warehouses (
    id INT PRIMARY KEY IDENTITY(1,1),
    name NVARCHAR(100) NOT NULL,
    address NVARCHAR(255),
    capacity INT,
    Latitude DECIMAL(9, 6),
    Longitude DECIMAL(10, 6)
);

-- 6. Bảng Tồn kho hiện tại (Inventory)
CREATE TABLE Inventory (
    id INT PRIMARY KEY IDENTITY(1,1),
    warehouse_id INT FOREIGN KEY REFERENCES Warehouses(id),
    product_id INT FOREIGN KEY REFERENCES Products(id),
    quantity INT NOT NULL DEFAULT 0,
    CONSTRAINT CHK_Inventory_Quantity CHECK (quantity >= 0),
    CONSTRAINT UQ_Warehouse_Product UNIQUE (warehouse_id, product_id) -- Đảm bảo 1 kho - 1 sản phẩm chỉ có 1 dòng
);

-- 7. Bảng Phiếu Nhập/Xuất (Receipts)
CREATE TABLE Receipts (
    id INT PRIMARY KEY IDENTITY(1,1),
    type NVARCHAR(10) CHECK (type IN ('INBOUND', 'OUTBOUND')),
    warehouse_id INT FOREIGN KEY REFERENCES Warehouses(id),
    staff_id INT FOREIGN KEY REFERENCES Users(id),
    partner_name NVARCHAR(255),
    created_at DATETIME DEFAULT GETDATE()
);

-- 8. Chi tiết phiếu Nhập/Xuất (Receipt_Details)
CREATE TABLE Receipt_Details (
    id INT PRIMARY KEY IDENTITY(1,1), -- Thêm ID tự tăng
    receipt_id INT FOREIGN KEY REFERENCES Receipts(id),
    product_id INT FOREIGN KEY REFERENCES Products(id),
    quantity INT NOT NULL CHECK (quantity > 0),
    price DECIMAL(18, 2),
    CONSTRAINT UQ_Receipt_Product UNIQUE (receipt_id, product_id) -- Tránh trùng sản phẩm trong 1 hóa đơn
);

-- 9. Lệnh điều chuyển (Transfer_Orders)
CREATE TABLE Transfer_Orders (
    id INT PRIMARY KEY IDENTITY(1,1),
    from_warehouse_id INT FOREIGN KEY REFERENCES Warehouses(id),
    to_warehouse_id INT FOREIGN KEY REFERENCES Warehouses(id),
    staff_id INT FOREIGN KEY REFERENCES Users(id),
    status NVARCHAR(20) DEFAULT 'PENDING' 
        CHECK (status IN ('PENDING', 'APPROVED', 'SHIPPING', 'COMPLETED', 'CANCELLED')),
    created_at DATETIME DEFAULT GETDATE()
);

-- 10. Chi tiết lệnh điều chuyển (Transfer_Details)
CREATE TABLE Transfer_Details (
    id INT PRIMARY KEY IDENTITY(1,1),
    transfer_id INT FOREIGN KEY REFERENCES Transfer_Orders(id),
    product_id INT FOREIGN KEY REFERENCES Products(id),
    quantity INT NOT NULL CHECK (quantity > 0),
    CONSTRAINT UQ_Transfer_Product UNIQUE (transfer_id, product_id)
);

-- 11. Thông tin giao hàng (Shipments)
CREATE TABLE Shipments (
    id INT PRIMARY KEY IDENTITY(1,1),
    transfer_id INT FOREIGN KEY REFERENCES Transfer_Orders(id),
    driver_name NVARCHAR(100),
    license_plate NVARCHAR(20),
    status NVARCHAR(20) DEFAULT 'PICKING'
        CHECK (status IN ('PICKING', 'SHIPPING', 'DELIVERED')),
    expected_delivery_at DATETIME,
    actual_delivery_at DATETIME
);

-- 12. Nhật ký tồn kho (Inventory_Logs)
CREATE TABLE Inventory_Logs (
    id INT PRIMARY KEY IDENTITY(1,1),
    product_id INT,
    warehouse_id INT,
    change_amount INT,
    action_type NVARCHAR(20),
    reference_id INT,
    created_at DATETIME DEFAULT GETDATE()
);
